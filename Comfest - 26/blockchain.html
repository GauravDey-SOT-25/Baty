// Parallel/contracts/TrustRegistry.sol
function addAdvancedTrustRecord(
    string memory urlHash,
    string memory certFingerprint,
    string memory ipfsCid,
    address[] memory quorumNodes,
    uint8 consensusScore,
    string memory metadata
) public onlyActiveValidator {
    require(bytes(urlHash).length > 0, "Empty urlHash");
    require(bytes(certFingerprint).length > 0, "Empty certFingerprint");

    if (!hashExists[urlHash]) {
        registeredHashes.push(urlHash);
        hashExists[urlHash] = true;
    }

    uint256 lockedCollateral = 1 ether;
    records[urlHash] = TrustRecord({
        urlHash: urlHash,
        certFingerprint: certFingerprint,
        ipfsCid: ipfsCid,
        timestamp: block.timestamp,
        primaryVerifier: msg.sender,
        oracleQuorum: quorumNodes,
        stakeLocked: lockedCollateral,
        isTrusted: true,
        consensusScore: consensusScore,
        metadata: metadata
    });

    emit LinkAttested(
        urlHash,
        urlHash,
        certFingerprint,
        ipfsCid,
        msg.sender,
        lockedCollateral,
        consensusScore,
        block.timestamp
    );
}
# Parallel/backend/app.py
@app.route("/api/check", methods=["GET"])
def api_check():
    """
    Takes a URL, hashes it, checks the smart contract, fetches live SSL certificate,
    and returns comprehensive Trust Score and verification verdict.
    Query parameter:
      - url: Target URL/domain to scan
    """
    raw_url = request.args.get("url", "").strip()
    if not raw_url:
        return jsonify({
            "success": False,
            "error": "Query parameter 'url' is required (e.g. /api/check?url=ethereum.org)"
        }), 400

    try:
        canonical_url, hostname, port = normalize_url(raw_url)
        url_hash = compute_url_hash(hostname)

        # 1. Query Blockchain Reader
        is_trusted_on_chain = blockchain.is_trusted(url_hash)
        chain_record = blockchain.get_trust_record(url_hash)

        # 2. Live SSL Certificate Inspection
        live_cert = fetch_ssl_fingerprint(hostname, port)

        # 3. Live Threat Intelligence (VirusTotal + Google Safe Browsing)
        threat_intel = get_threat_intelligence(canonical_url, hostname)

        # 4. Trust Scoring Engine
        verdict = evaluate_trust(
            url_hash=url_hash,
            chain_record=chain_record,
            is_trusted_on_chain=is_trusted_on_chain,
            live_cert=live_cert,
            threat_intel=threat_intel
        )

# Parallel/backend/hasher.py
def compute_url_hash(input_str: str) -> str:
    """
    Computes standard SHA-256 hash of the canonical domain/URL.
    Returns 0x-prefixed 64-character lowercase hex string.
    """
    _, host, _ = normalize_url(input_str)
    # Hash the normalized domain name
    raw_hash = hashlib.sha256(host.encode("utf-8")).hexdigest()
    return f"0x{raw_hash}"

FAQ

About the prototype.

Does every result come from a public blockchain?

No. The backend can use a configured local Ethereum RPC, or run in a standalone in-memory ledger mode. Check `/api/status` to see the current mode.

Does a verified result guarantee a site is safe?

No. It means the prototype's available record and certificate checks match and no configured threat source returned a flag. New or unreported attacks may not appear in those signals.

What happens when no record exists?

The current scoring function returns “Unverified” with a score of zero. It does not treat an unknown domain as trusted by default.

Can trust be revoked?

The smart contract exposes an owner-only slashing path. The standalone backend's in-memory ledger has a simulated revocation operation.
